class FunctionAnalyzer:
    """Phase 1: Analyzes input functions and classifies them"""
    
    def __init__(self):
        self.z = sp.Symbol('z', complex=True)
        self.x = sp.Symbol('x', real=True)
        
    def parse_input(self, function_str: str) -> Dict[str, Any]:
        """
        Algorithm 1: Input Parsing and Initialization
        """
        print("\n" + "="*60)
        print("PHASE 1: FUNCTION AND INTEGRAL ANALYSIS")
        print("="*60)
        
        # Replace x with z for complex analysis
        function_of_z_str = function_str.replace('x', 'z')
        f_z = sp.sympify(function_of_z_str)
        
        print(f"\n[Input Parsing]")
        print(f"  Original function: {function_str}")
        print(f"  Complex function: {f_z}")
        
        return {
            'original': function_str,
            'complex_form': f_z,
            'variable': self.z
        }
    
    def classify_function(self, f_z) -> Dict[str, Any]:
        """
        Function Classification Decision Point
        
        Returns type classification and strategy
        """
        print(f"\n[Function Classification]")
        
        # Check for trigonometric functions
        trig_funcs = f_z.atoms(sp.cos, sp.sin)
        
        if trig_funcs:
            # Type 2: Rational with Trigonometry
            trig_func = trig_funcs.pop()
            rational_part = f_z / trig_func
            
            numer, denom = sp.fraction(rational_part)
            deg_diff = sp.degree(denom, self.z) - sp.degree(numer, self.z)
            
            print(f"  Type: Rational Function with Trigonometry")
            print(f"  Trigonometric component: {trig_func}")
            print(f"  Rational part: {rational_part}")
            print(f"  Degree condition: deg(Q) - deg(P) = {deg_diff}")
            print(f"  Strategy: Jordan's Lemma")
            
            return {
                'type': 'TYPE_2_TRIG',
                'trig_func': trig_func,
                'rational_part': rational_part,
                'strategy': 'jordan_lemma'
            }
        else:
            # Type 1: Pure Rational
            numer, denom = sp.fraction(f_z)
            deg_diff = sp.degree(denom, self.z) - sp.degree(numer, self.z)
            
            print(f"  Type: Pure Rational Function")
            print(f"  Numerator degree: {sp.degree(numer, self.z)}")
            print(f"  Denominator degree: {sp.degree(denom, self.z)}")
            print(f"  Degree condition: deg(Q) - deg(P) = {deg_diff}")
            
            if deg_diff >= 2:
                print(f"  ✓ Convergence condition satisfied (≥2)")
                print(f"  Strategy: Standard Semicircular Contour")
            else:
                print(f"  ✗ Warning: May not converge at infinity")
            
            return {
                'type': 'TYPE_1_RATIONAL',
                'rational_part': f_z,
                'strategy': 'semicircle',
                'degree_diff': deg_diff
            }
    
    def detect_and_classify_poles(self, f_z) -> List[Dict[str, Any]]:
        """
        Algorithm 2: Pole Detection and Classification (Robust Version)
        
        Uses numeric root-finding (nroots) first for stability, with fallback to solve().
        Handles symbolic poles, RootOf objects, and complex conversion safely.
        """
        print(f"\n[Pole Detection & Classification]")
        
        # Extract denominator
        denominator = sp.fraction(f_z)[1]
        print(f"  Denominator: {denominator}")
        
        # Try numeric root-finding first for robust numeric values
        poles_raw = []
        try:
            # nroots returns numeric approximations (mpmath/complex numbers)
            poles_raw = list(sp.nroots(denominator, n=15))  # 15 digits precision
            print("  ✓ Found roots using nroots() [numeric approximation]")
        except Exception as e:
            # Fallback to symbolic solve
            try:
                poles_raw = sp.solve(denominator, self.z)
                print("  ✓ Found roots using solve() [symbolic]")
            except Exception as solve_err:
                print(f"  ✗ Error finding roots: {solve_err}")
                return []
        
        if not poles_raw:
            print("  ⚠️ No poles found")
            return []
        
        print(f"  All poles found (raw): {poles_raw}")
        
        # Get multiplicities for accurate order detection
        poles_dict = {}
        try:
            poles_dict = sp.roots(denominator, self.z, multiple=False)
        except Exception:
            # If roots() fails, assume all poles are simple
            poles_dict = {p: 1 for p in poles_raw}
        
        poles_info = []
        eps = 1e-8  # Tolerance for real axis detection
        
        for pole in poles_raw:
            # Convert to numeric complex if possible
            p_complex = None
            real_part = None
            imag_part = None
            
            try:
                # Attempt to get numeric value
                if hasattr(pole, 'evalf'):
                    peval = pole.evalf()
                    p_complex = complex(peval)
                else:
                    p_complex = complex(pole)
                
                real_part = float(p_complex.real)
                imag_part = float(p_complex.imag)
                
            except (TypeError, ValueError, AttributeError) as e:
                # Pole is symbolic or cannot be converted
                try:
                    # Second attempt: direct complex conversion
                    p_complex = complex(pole)
                    real_part = float(p_complex.real)
                    imag_part = float(p_complex.imag)
                except Exception:
                    print(f"  ⚠️ Warning: Could not convert pole {pole} to numeric. Storing symbolically.")
                    p_complex = pole  # Keep as symbolic
            
            # Classify by location
            if real_part is None or imag_part is None:
                location = "Symbolic/Unknown"
                enclosed = False
            elif imag_part > eps:
                location = "Upper Half-Plane (UHP)"
                enclosed = True
            elif imag_part < -eps:
                location = "Lower Half-Plane (LHP)"
                enclosed = False
            else:
                location = "Real Axis"
                enclosed = "Special"
            
            # Get multiplicity (order)
            order = poles_dict.get(pole, 1)
            if not isinstance(order, int) or order < 1:
                order = 1
            
            pole_type = "Simple" if order == 1 else f"Multiple (order {order})"
            
            pole_info = {
                'pole': pole,
                'value': p_complex if p_complex is not None else pole,
                'real': real_part,
                'imag': imag_part,
                'order': order,
                'type': pole_type,
                'location': location,
                'enclosed': enclosed
            }
            
            poles_info.append(pole_info)
            
            # Display pole information
            display_value = p_complex if p_complex is not None else pole
            print(f"  • Pole: {display_value}")
            print(f"    Location: {location}")
            print(f"    Type: {pole_type}")
            print(f"    Enclosed in UHP contour: {enclosed}")
        
        return poles_info

class ContourSelector:
    """Phase 2: Selects optimal contour based on analysis"""
    
    @staticmethod
    def select_contour(classification: Dict, poles_info: List[Dict]) -> Dict[str, Any]:
        """
        Contour selection decision tree
        """
        print("\n" + "="*60)
        print("PHASE 2: CONTOUR SELECTION & STRATEGY")
        print("="*60)
        
        func_type = classification['type']
        
        # Filter poles in upper half-plane
        uhp_poles = [p for p in poles_info if p['enclosed'] == True]
        real_poles = [p for p in poles_info if p['enclosed'] == "Special"]
        
        print(f"\n[Contour Decision]")
        print(f"  Function Type: {func_type}")
        print(f"  Poles in UHP: {len(uhp_poles)}")
        print(f"  Poles on Real Axis: {len(real_poles)}")
        
        if func_type == 'TYPE_1_RATIONAL':
            return ContourSelector._type1_contour(uhp_poles, real_poles)
        elif func_type == 'TYPE_2_TRIG':
            return ContourSelector._type2_contour(uhp_poles, real_poles)
        else:
            return ContourSelector._type3_contour(poles_info)
    
    @staticmethod
    def _type1_contour(uhp_poles: List, real_poles: List) -> Dict:
        """Type 1: Standard semicircular contour"""
        print(f"\n  → Type 1 Strategy Selected")
        print(f"  Contour: Semicircular (D-shape) in UHP")
        print(f"  Justification: Integral over arc vanishes as R→∞")
        print(f"  Action: Sum residues of {len(uhp_poles)} poles in UHP")
        
        if real_poles:
            print(f"  ⚠ Special handling: {len(real_poles)} poles on real axis")
            print(f"  Using indented contour around real poles")
        
        return {
            'type': 'semicircle_uhp',
            'poles_to_process': uhp_poles,
            'special_poles': real_poles,
            'method': 'standard_residue'
        }
    
    @staticmethod
    def _type2_contour(uhp_poles: List, real_poles: List) -> Dict:
        """Type 2: Jordan's lemma application"""
        print(f"\n  → Type 2 Strategy Selected (Jordan's Lemma)")
        print(f"  Contour: Semicircular in UHP")
        print(f"  Justification: Jordan's lemma for exponential decay")
        print(f"  Action: Process {len(uhp_poles)} UHP poles, extract Re/Im")
        
        return {
            'type': 'semicircle_uhp_jordan',
            'poles_to_process': uhp_poles,
            'special_poles': real_poles,
            'method': 'jordan_lemma'
        }
    
    @staticmethod
    def _type3_contour(poles_info: List) -> Dict:
        """Type 3: Unit circle"""
        print(f"\n  → Type 3 Strategy Selected")
        print(f"  Contour: Unit Circle |z|=1")
        print(f"  Justification: Perfect mapping of [0,2π]")
        print(f"  Action: Transform and sum residues inside unit circle")
        
        inside_unit = [p for p in poles_info if p['real'] is not None and abs(p['value']) < 1]
        
        return {
            'type': 'unit_circle',
            'poles_to_process': inside_unit,
            'method': 'unit_circle_substitution'
        }

class ResidueCalculator:
    """Phase 3: Computes residues and final solution"""
    
    def __init__(self):
        self.z = sp.Symbol('z', complex=True)
    
    def calculate_residue(self, func, pole_info: Dict) -> Any:
        """
        Algorithm 3: Residue Calculation
        """
        pole = pole_info['pole']
        order = pole_info['order']
        
        try:
            if order == 1:
                # Simple pole formula
                residue = sp.limit((self.z - pole) * func, self.z, pole)
            else:
                # Multiple pole formula
                expr = (self.z - pole)**order * func
                for _ in range(order - 1):
                    expr = sp.diff(expr, self.z)
                residue = sp.limit(expr, self.z, pole) / sp.factorial(order - 1)
            
            return sp.simplify(residue)
        
        except Exception as e:
            print(f"  ⚠️ Error calculating residue at {pole}: {e}")
            return 0
    
    def apply_residue_theorem(self, residues: List, method: str, 
                            trig_func=None) -> Any:
        """
        Apply Residue Theorem: ∮f(z)dz = 2πi Σ Res(f,a_k)
        """
        print("\n[Applying Residue Theorem]")
        
        total_residue = sum(residues)
        print(f"  Sum of residues: {total_residue}")
        
        complex_result = 2 * sp.pi * sp.I * total_residue
        print(f"  Complex integral: 2πi × Σ residues = {complex_result}")
        
        if method == 'jordan_lemma' and trig_func:
            if isinstance(trig_func, sp.cos):
                result = sp.re(complex_result)
                print(f"  Taking REAL part for cos")
            elif isinstance(trig_func, sp.sin):
                result = sp.im(complex_result)
                print(f"  Taking IMAGINARY part for sin")
            else:
                result = complex_result
        else:
            result = complex_result
        
        return sp.simplify(result)

class ContourIntegrationSystem:
    """
    Complete system integrating all three phases
    """
    
    def __init__(self):
        self.analyzer = FunctionAnalyzer()
        self.selector = ContourSelector()
        self.calculator = ResidueCalculator()
    
    def evaluate_integral(self, function_str: str, silent: bool = False) -> Dict[str, Any]:
        """
        Algorithm 4: Complete Inverse Transform Computation
        """
        start_time = time.time()
        
        # PHASE 1: Analysis
        parsed = self.analyzer.parse_input(function_str)
        f_z = parsed['complex_form']
        
        classification = self.analyzer.classify_function(f_z)
        poles_info = self.analyzer.detect_and_classify_poles(
            classification.get('rational_part', f_z)
        )
        
        # PHASE 2: Contour Selection
        contour_plan = self.selector.select_contour(classification, poles_info)
        
        # PHASE 3: Calculate Solution
        if not silent:
            print("\n" + "="*60)
            print("PHASE 3: CALCULATION & SOLUTION")
            print("="*60)
            print("\n[Calculate Residues]")
        
        residues = []
        
        # Prepare function for residue calculation (ROBUST TRIG HANDLING)
        if classification['type'] == 'TYPE_2_TRIG':
            trig_func = classification['trig_func']
            inner = trig_func.args[0]  # typically a*z or z
            
            # Extract frequency coefficient 'a' such that inner = a*z
            a = None
            if inner.has(self.analyzer.z):
                try:
                    # Divide by z to get coefficient
                    a_expr = sp.simplify(inner / self.analyzer.z)
                    # Try to convert to numeric
                    try:
                        a_val = complex(a_expr.evalf())
                        a = a_val
                        if not silent:
                            print(f"  Extracted frequency coefficient: a = {a}")
                    except Exception:
                        # Keep symbolic if conversion fails
                        a = a_expr
                        if not silent:
                            print(f"  Frequency coefficient (symbolic): a = {a}")
                except Exception:
                    a = 1  # Default fallback
                    if not silent:
                        print(f"  Using default coefficient: a = 1")
            else:
                # No z present: treat inner as constant frequency
                try:
                    a = complex(inner.evalf())
                    if not silent:
                        print(f"  Constant frequency: a = {a}")
                except Exception:
                    a = inner
                    if not silent:
                        print(f"  Symbolic frequency: a = {a}")
            
            # Build exponential factor e^{iaz}
            if isinstance(a, (int, float, complex)):
                g = classification['rational_part'] * sp.exp(sp.I * a * self.calculator.z)
            else:
                g = classification['rational_part'] * sp.exp(sp.I * a * self.calculator.z)
            
            func_to_analyze = g
            if not silent:
                print(f"  Analyzing function: {func_to_analyze}")
        else:
            func_to_analyze = f_z
            trig_func = None
        
        # Calculate residues for each pole
        for pole_info in contour_plan['poles_to_process']:
            if not silent:
                print(f"\n  Processing pole: {pole_info['value']}")
            res = self.calculator.calculate_residue(func_to_analyze, pole_info)
            if not silent:
                print(f"  Residue: {res}")
            residues.append(res)
        
        # Apply Residue Theorem
        result = self.calculator.apply_residue_theorem(
            residues, 
            contour_plan['method'],
            trig_func
        )
        
        computation_time = (time.time() - start_time) * 1000  # ms
        
        if not silent:
            print("\n" + "="*60)
            print("FINAL RESULT")
            print("="*60)
            print(f"\n  Integral value: {result}")
        
        # Safe numerical evaluation
        try:
            numerical = complex(result.evalf())
            if not silent:
                print(f"  Numerical value: {numerical}")
        except Exception:
            numerical = result
            if not silent:
                print(f"  Numerical value: {result} (symbolic)")
        
        if not silent:
            print(f"  Computation time: {computation_time:.3f} ms")
        
        return {
            'result': result,
            'numerical': numerical,
            'time_ms': computation_time,
            'poles_processed': len(residues),
            'classification': classification['type']
        }

    def visualize_contour(self, function_str: str, save_path: Optional[str] = None):
        """
        Generate complex plane visualization with SAFE numeric conversion
        """
        parsed = self.analyzer.parse_input(function_str)
        f_z = parsed['complex_form']
        
        classification = self.analyzer.classify_function(f_z)
        poles_info = self.analyzer.detect_and_classify_poles(
            classification.get('rational_part', f_z)
        )
        
        # Helper function for safe conversion to float
        def _to_float_parts(p):
            """Convert pole to (real, imag) floats safely"""
            try:
                if isinstance(p, (int, float, complex)):
                    return float(np.real(p)), float(np.imag(p))
                elif hasattr(p, 'evalf'):
                    pc = complex(p.evalf())
                    return float(pc.real), float(pc.imag)
                else:
                    pc = complex(p)
                    return float(pc.real), float(pc.imag)
            except Exception:
                # Fallback to origin for plotting
                print(f"  Warning: Could not plot pole {p}, using (0,0)")
                return 0.0, 0.0
        
        # Extract pole values
        all_pole_values = [p['value'] for p in poles_info if p['value'] is not None]
        uhp_pole_values = [p['value'] for p in poles_info if p['enclosed'] == True]
        
        # Convert to plottable coordinates
        pole_reals = []
        pole_imags = []
        for p in all_pole_values:
            r, im = _to_float_parts(p)
            pole_reals.append(r)
            pole_imags.append(im)
        
        uhp_reals = []
        uhp_imags = []
        for p in uhp_pole_values:
            r, im = _to_float_parts(p)
            uhp_reals.append(r)
            uhp_imags.append(im)
        
        # Plot setup
        fig, ax = plt.subplots(figsize=(10, 8))
        
        # Determine plot limits - FIXED: Ensure float type
        if pole_reals and pole_imags:
            max_real = max(abs(r) for r in pole_reals)
            max_imag = max(abs(i) for i in pole_imags)
            max_val = float(max(max_real, max_imag, 2) * 1.3)
        else:
            max_val = 3.0
        
        # Plot semicircular contour
        ax.plot([-max_val, max_val], [0.0, 0.0], 'b-', linewidth=2, label='Real Axis')
        arc_theta = np.linspace(0, np.pi, 200)
        arc_x = max_val * np.cos(arc_theta)
        arc_y = max_val * np.sin(arc_theta)
        ax.plot(arc_x, arc_y, 'b-', linewidth=2, label='Semicircular Arc')
        
        # Plot all poles
        if pole_reals:
            ax.scatter(pole_reals, pole_imags, marker='x', color='black', 
                      s=200, linewidths=3, label='All Poles', zorder=5)
        
        # Highlight UHP poles
        if uhp_reals:
            ax.scatter(uhp_reals, uhp_imags, marker='o', facecolors='none',
                      edgecolors='red', s=400, linewidths=3, 
                      label='Poles in UHP', zorder=6)
        
        # Formatting
        ax.axhline(0, color='black', linewidth=0.5)
        ax.axvline(0, color='black', linewidth=0.5)
        ax.set_xlabel("Re(z)", fontsize=14)
        ax.set_ylabel("Im(z)", fontsize=14)
        ax.set_title(f"Complex Plane: {function_str}", fontsize=16, fontweight='bold')
        ax.grid(True, linestyle='--', alpha=0.6)
        ax.set_aspect('equal', 'box')
        ax.legend(fontsize=12, loc='upper right')
        
        # Set limits - FIXED: Use separate float arguments
        x_limit = max_val * 1.1
        y_limit_low = -max_val * 0.3
        y_limit_high = max_val * 1.1
        
        ax.set_xlim(-x_limit, x_limit)
        ax.set_ylim(y_limit_low, y_limit_high)
        
        plt.tight_layout()
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"✓ Figure saved to {save_path}")
        
        plt.show()

def run_test_suite(generate_viz: bool = True):
    """
    Execute all test cases from the research paper
    """
    system = ContourIntegrationSystem()
    
    test_cases = [
        {
            'name': 'Test Case 1: Second-Order System',
            'function': '1/(z**2 + 2*z + 5)',
            'description': 'Underdamped filter response'
        },
        {
            'name': 'Test Case 2: Digital Filter',
            'function': 'z/((z-0.5)*(z-0.8))',
            'description': 'IIR filter impulse response'
        },
        {
            'name': 'Test Case 3: High-Order System',
            'function': '1/((z+1)*(z+2)*(z**2+2*z+5))',
            'description': 'Fourth-order filter cascade'
        },
        {
            'name': 'Test Case 4: Fourier-Type',
            'function': 'cos(z)/(z**2+1)',
            'description': 'Jordan\'s lemma application'
        },
        {
            'name': 'Test Case 5: Simple Rational',
            'function': '1/(z**2+4)',
            'description': 'Basic rational function'
        }
    ]
    
    results = []
    
    for i, test in enumerate(test_cases, 1):
        print(f"\n{'#'*70}")
        print(f"{'#'*70}")
        print(f"{test['name']}")
        print(f"Description: {test['description']}")
        print(f"{'#'*70}")
        
        try:
            result = system.evaluate_integral(test['function'])
            
            results.append({
                'Test Case': test['name'],
                'Function': test['function'],
                'Result': str(result['numerical']),
                'Time (ms)': f"{result['time_ms']:.3f}",
                'Poles': result['poles_processed'],
                'Type': result['classification']
            })
            
            # Generate visualization for first 3 cases
            if generate_viz and i <= 3:
                print(f"\nGenerating visualization...")
                viz_path = f'../paper/figures/test_case_{i}_contour.png'
                os.makedirs('../paper/figures', exist_ok=True)
                system.visualize_contour(test['function'], save_path=viz_path)
        
        except Exception as e:
            print(f"\n✗ Error in test case: {e}")
            results.append({
                'Test Case': test['name'],
                'Function': test['function'],
                'Result': f'ERROR: {str(e)[:50]}',
                'Time (ms)': 'N/A',
                'Poles': 'N/A',
                'Type': 'N/A'
            })
    
    # Generate performance table
    print("\n" + "="*70)
    print("PERFORMANCE SUMMARY TABLE")
    print("="*70)
    df = pd.DataFrame(results)
    print(df.to_string(index=False))
    
    # Save to CSV
    os.makedirs('../data', exist_ok=True)
    df.to_csv('../data/test_results.csv', index=False)
    print("\n✓ Results saved to '../data/test_results.csv'")
    
    return df

def interactive_mode():
    """
    Interactive calculator interface
    """
    system = ContourIntegrationSystem()
    
    print("="*70)
    print("CONTOUR INTEGRATION CALCULATOR")
    print("Optimal Contour Selection for Inverse Transforms")
    print("="*70)
    print("\nExamples:")
    print("  • 1/(x**2 + 4)")
    print("  • cos(x)/(x**2 + 1)")
    print("  • z/((z-0.5)*(z-0.8))")
    print("\nCommands:")
    print("  'quit' or 'exit' - Exit program")
    print("  'viz FUNCTION' - Visualize contour for FUNCTION")
    print("  'test' - Run test suite")
    print("  'figures' - Generate paper figures")
    print("  'export' - Export data files")
    print("="*70)
    
    while True:
        try:
            function_str = input("\nEnter function to integrate: ").strip()
            
            if function_str.lower() in ['exit', 'quit']:
                print("Exiting calculator. Goodbye!")
                break
            
            if function_str.lower() == 'test':
                run_test_suite()
                continue
            
            if function_str.lower() == 'figures':
                generate_performance_figures()
                continue
            
            if function_str.lower() == 'export':
                export_test_results()
                continue
            
            if not function_str:
                continue
            
            if function_str.lower().startswith('viz '):
                func = function_str[4:].strip()
                system.visualize_contour(func)
            else:
                system.evaluate_integral(function_str)
        
        except KeyboardInterrupt:
            print("\n\nExiting calculator. Goodbye!")
            break
        except Exception as e:
            print(f"\n✗ Error: Could not process function")
            print(f"Details: {e}")
            import traceback
            traceback.print_exc()
